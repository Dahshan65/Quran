function showStageResult() { const start = stage * 10 + 1; const end = Math.min((stage + 1) * 10, questions.length); const count = end - start + 1; const percentage = Math.round((lastStageScore / count) * 100); const passed = percentage > 70; $("stageResultTitle").textContent = `نتيجة المرحلة ${stage + 1}`; $("stageResultScore").textContent = `${lastStageScore} / ${count} (${percentage}%)`; $("stageResultText").textContent = passed ? "أحسنت، بارك الله فيك" : "أعد المحاولة واستمر في الحفظ"; $("continueButton").textContent = passed ? (end >= questions.length ? "عرض النتيجة النهائية" : "الانتقال إلى المرحلة التالية") : "إعادة المرحلة"; $("continueButton").onclick = passed ? continueAfterStage : retryStage; show("stageResult"); } function retryStage() { // إعادة أسئلة المرحلة الحالية فقط const start = stage * 10; const end = Math.min(start + 10, questions.length); questions.splice(start, end - start, ...shuffle(questions.slice(start, end))); index = start; stageCorrect = 0; lastStageScore = 0; lastStageFinished = false; $("nextButton").dataset.answered = "no"; renderStages(); show("quiz"); renderQuestion(); } function continueAfterStage() { if (!lastStageFinished) return; const count = Math.min((stage + 1) * 10, questions.length); const percentage = Math.round((lastStageScore / (count - stage * 10)) * 100); // لا انتقال إلى المرحلة التالية دون تجاوز 70% if (percentage <= 70) { retryStage(); return; } if (index + 1 >= questions.length) { showFinal(); return; } index++; stage = Math.floor(index / 10); stageCorrect = 0; stageStart = index; lastStageFinished = false; renderStages(); show("quiz"); renderQuestion(); }
